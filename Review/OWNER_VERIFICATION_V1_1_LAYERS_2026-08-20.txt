  entities = model.active_entities

  # Add a few V1.1-specific layer fixtures. SU locks the default
  # 'Layer0' as the construction default (we will classify it
  # automatically), so adding new named layers is fine.
  layer_dim   = layers.add('DIM-WALLS')    # :dimension role
  layer_anno  = layers.add('TXT-LABELS')    # :annotation role
  layer_guide = layers.add('GUIDE-XLINES')  # :guide role
  layer_oops  = layers.add('XYZ-9999')      # :unknown role (no rule match)

  # A hidden layer (verify muted style + sort-bottom-in-bucket).
  # Owner Gate 2 V1.1 NIT (corrected): the name MUST match the
  # Annotation regex `/(anno|t?ext|label)/i` so it is classified
  # as :annotation rather than falling through to :unknown. The
  # original 'TXT-HIDDEN' did not match and was relabeled to
  # 'TXT-LABEL-HIDDEN' (matches via the `label` alternation).
  layer_hidden = layers.add('TXT-LABEL-HIDDEN')
  layer_hidden.visible = false  # OFF-SCREEN

  # Helper: build a 4-edge closed rectangle whose edges live on a
  # NAMED LAYER. Owner Gate 2 V1.1 NIT (corrected):
  #   - Named layers must be assigned to the Edge objects
  #     themselves, not only to the parent Group. Relying on group
  #     inheritance can leave edges on Layer0 in some SketchUp
  #     builds; explicit per-edge assignment is unambiguous and
  #     works on all SU2017+ versions.
  #   - Closed rectangles (4 edges forming a loop) are the right
  #     '0 issues' fixture. A single edge produces 2 Open Endpoint
  #     issues, so it cannot represent an issue-free layer. Closed
  #     rectangles produce 0 Open Endpoints, 0 Short Edges, 0
  #     Duplicates, and 0 Gap Candidates.
  def make_rect(entities, layer_obj, x, y)
    g = entities.add_group
    e1 = g.entities.add_line([x,    y,    0], [x+10, y,    0])
    e2 = g.entities.add_line([x+10, y,    0], [x+10, y+5,  0])
    e3 = g.entities.add_line([x+10, y+5,  0], [x,    y+5,  0])
    e4 = g.entities.add_line([x,    y+5,  0], [x,    y,    0])
    [e1, e2, e3, e4].each { |e| e.layer = layer_obj }
    g
  end

  # Fixture 1 (V1.0 baseline): 4-edge closed square on Layer0.
  test_group = entities.add_group
  test_group.layer = layers['Layer0']
  test_group.entities.add_line([0,0,0],      [10,0,0])
  test_group.entities.add_line([10,0,0],     [10,5,0])
  test_group.entities.add_line([10,5,0],     [0,5,0])
  test_group.entities.add_line([0,5,0],      [0,0,0])

  # Fixture 2 (V1.1): closed rectangle on the DIM-WALLS layer

Capture PRE-analysis Ruby fingerprint (paste once, then re-paste

  # -- Begin V1.1 fingerprint helper (extends V1.0 v6.2) --
  def su_count_geometry_recursive(entities)
    counts = { edges: 0, faces: 0, groups: 0,
               components: 0, definitions: 0 }
    entities.each do |e|
      case e
      when Sketchup::Edge
        counts[:edges] += 1
      when Sketchup::Face
        counts[:faces] += 1
      when Sketchup::Group
        counts[:groups] += 1
        counts.merge!(su_count_geometry_recursive(e.entities)) { |_, a, b| a + b }
      when Sketchup::ComponentInstance
        counts[:components] += 1
        counts.merge!(su_count_geometry_recursive(e.definition.entities)) { |_, a, b| a + b }
      end
    end
    counts
  end

  def su_active_path_size_safe(model)
    return 0 unless model.respond_to?(:active_path)
    ap = model.active_path
    ap.nil? ? 0 : ap.length
  end

  def su_visible_layers(model)
    model.layers.map { |l| [l.name, l.visible?] }
  end

  def su_fingerprint_v11(model)
    {
      edges:           su_count_geometry_recursive(model.entities)[:edges],
      groups:          su_count_geometry_recursive(model.entities)[:groups],
      layers_visible:  su_visible_layers(model),
      active_path_size: su_active_path_size_safe(model)
    }
  end

  fp_before = su_fingerprint_v11(Sketchup.active_model)
  puts "PRE  = #{fp_before.inspect}"
  # -- End fingerprint helper --
