    # -- Begin fingerprint helper v6.2 (recursive geometry/count) --
    def su_count_geometry_recursive(entities)
      counts = { edges: 0, faces: 0, groups: 0,
                 components: 0, definitions: 0 }
      entities.each do |e|
        case e
        when Sketchup::Edge
          counts[:edges] += 1
        when Sketchup::Face
          counts[:faces] += 1
        when Sketchup::Group
          counts[:groups] += 1
          counts.merge!(su_count_geometry_recursive(e.entities)) { |_, a, b| a + b }
        when Sketchup::ComponentInstance
          counts[:components] += 1
          counts.merge!(su_count_geometry_recursive(e.definition.entities)) { |_, a, b| a + b }
        end
      end
      counts
    end

    def su_active_path_size_safe(model)
      return 0 unless model.respond_to?(:active_path)
      ap = model.active_path
      ap.nil? ? 0 : ap.length
    end

    # GEOMETRY/COUNT fingerprint only. This fingerprint detects
    # COUNT changes (entities added or removed). It does NOT prove
    # absence of property-only mutations (material / layer / hidden /
    # soft / dictionary entries). For property-only mutations, the
    # test_no_overlay_lint + real-host behavior evidence are
    # complementary guards (see step N.6).
    def su_fingerprint_v6(model)
      m = model
      geom = su_count_geometry_recursive(m.entities)
      {
        edges:           geom[:edges],
        faces:           geom[:faces],
        groups:          geom[:groups],
        components:      geom[:components],
        definitions:     m.definitions.length,
        layers:          m.layers.length,
        materials:       m.materials.length,
        styles:          m.styles.length,
        tags:            (m.respond_to?(:tags) ? m.tags.length : 0),
        active_path_size: su_active_path_size_safe(m)
      }
    end

    fp_before = su_fingerprint_v6(Sketchup.active_model)
    puts "PRE  = #{fp_before.inspect}"
    # -- End fingerprint helper --
