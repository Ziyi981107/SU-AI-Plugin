  model = Sketchup.active_model
  entities = model.active_entities
  layers = model.layers

  # V13-1: one rectangular face on Layer0 (4 vertices, 0 holes).
  # SketchUp's `entities.add_face` is the easiest path to a
  # rectangular face. Build a 4-vertex outer loop:
  v1 = entities.add_face(
    [0,  0,  0],
    [10, 0,  0],
    [10, 5,  0],
    [0,  5,  0]
  )
  # v1's layer defaults to Layer0 (the model default).

  # V13-2: one face explicitly assigned to DIM-WALLS.
  layer_dim = layers.add('DIM-WALLS')
  v2_group = entities.add_group
  v2_group.layer = layer_dim
  v2 = v2_group.entities.add_face(
    [20, 0,  0],
    [40, 0,  0],
    [40, 5,  0],
    [20, 5,  0]
  )

  # V13-3: one face with an inner loop / hole. Real SketchUp
  # creates a face with an inner loop when you push inner edges
  # inward through a vertex split. This is fragile on different
  # SU versions; if SU fails to create the inner loop on your
  # build, set v3_has_holes = false and proceed.
  layer_holes = layers.add('HOLES-OK')
  v3_group = entities.add_group
  v3_group.layer = layer_holes
  v3_outer = v3_group.entities.add_face(
    [60, 0,  0],
    [80, 0,  0],
    [80, 20, 0],
    [60, 20, 0]
  )
  v3_has_holes = false
  begin
    # Attempt to push an inner triangle through v3_outer to create
    # a face with one inner loop. This is a SketchUp API call
    # that may not produce a hole on every SU version.
    inner = v3_group.entities.add_face(
      [66,  5,  0],
      [74,  5,  0],
      [70, 12,  0]
    )
    inner.reverse!  # ensure opposite winding -> creates a hole
    v3_has_holes = (inner.is_a?(Sketchup::Face))
  rescue StandardError
    v3_has_holes = false
  end

  # V13-4: the same ComponentDefinition instantiated twice.
  # Build a face inside a component definition, then instantiate
  # the component twice in the model root.
  comp_defn = model.definitions.add('V13Comp')
  comp_defn.entities.add_face(
    [100, 0,  0],
    [110, 0,  0],
    [110, 5,  0],
    [100, 5,  0]
  )
  inst_a = entities.add_instance(
    comp_defn, Geom::Transformation.new([100, 0, 0])
  )
  inst_b = entities.add_instance(
    comp_defn, Geom::Transformation.new([130, 0, 0])
  )

  # V13-5: hidden layer + unknown-role presentation.

  v5_group.layer = layer_hidden

    [160, 5,  0]

  # PRE-analysis fingerprint helper (mirror V1.1 v11 / V1.2 v12):
  # -- Begin V1.3 fingerprint helper --
  def v13_count_geometry_recursive(entities)
    counts = { edges: 0, faces: 0, groups: 0,
               components: 0, definitions: 0 }
    entities.each do |e|
      case e
      when Sketchup::Edge
        counts[:edges] += 1
      when Sketchup::Face
        counts[:faces] += 1
      when Sketchup::Group
        counts[:groups] += 1
        counts.merge!(v13_count_geometry_recursive(e.entities)) { |_, a, b| a + b }
      when Sketchup::ComponentInstance
        counts[:components] += 1
        counts.merge!(v13_count_geometry_recursive(e.definition.entities)) { |_, a, b| a + b }
      end
    end
    counts
  end

  def v13_face_count(model)
    n = 0
    v13_count_geometry_recursive(model.entities)
    v13_count_geometry_recursive(model.entities)
  end

  def v13_total_faces(model)
    n = 0
    stack = [model.entities]
    until stack.empty?
      coll = stack.pop
      coll.each do |e|
        case e
        when Sketchup::Face
          n += 1
        when Sketchup::Group
          stack.push(e.entities)
        when Sketchup::ComponentInstance
          stack.push(e.definition.entities)
        end
      end
    end
    n
  end

  def v13_fingerprint(model)
    {
      edges:           v13_count_geometry_recursive(model.entities)[:edges],
      faces:           v13_total_faces(model),
      groups:          v13_count_geometry_recursive(model.entities)[:groups],
      components:      v13_count_geometry_recursive(model.entities)[:components],
      layers_visible:  model.layers.map { |l| [l.name, l.visible?] }
    }
  end

  fp_before = v13_fingerprint(Sketchup.active_model)
  puts "PRE = #{fp_before.inspect}"
  # -- End fingerprint helper --
